from odoo import models, fields, api


class Pieza(models.Model):
    _name = 'taller_mecanico.pieza'
    _description = 'Modelo para las pieza'

    nombre = fields.Char(String="Nombre")
    codigo = fields.Integer(String="Código")
    precio_unitario = fields.Integer(String="Precio unitario")

    reparacion_ids = fields.Many2many('taller_mecanico.reparacion', string='Reparaciones donde se usó')

    @api.constrains('precio_unitario')
    def comprobar_precio(self):
        for record in self:
            if record.precio_unitario < 0:
                raise ValidationError(f"El precio unitario debe ser mayor de 0")
```

# -*- coding: utf-8 -*-

from odoo import models, fields, api


class Reparacion(models.Model):
    _name = 'taller_mecanico.reparacion'
    _description = 'Modelo para las reparaciones'

    fecha_inicio = fields.Date(String="Fecha de inicio")
    fecha_fin = fields.Date(String="Fecha fin")
    descripcion = fields.Char()
    estado = fields.Selection([
        ('borrador', 'Borrador'),
        ('encurso', 'En curso'),
        ('finalizada', 'Finalizada')
    ], String='Estado')
    coste_total = fields.Integer()


    vehiculo_id = fields.Many2one('taller_mecanico.vehiculo', string='Vehículo')
    tecnico_id = fields.Many2one('res.users', string='Técnico')
    pieza_ids = fields.Many2many('taller_mecanico.pieza', string='Piezas utilizadas')

    def action_cambiar_borrador(self):
        for record in self:
            record.estado = 'borrador';
    def action_cambiar_encurso(self):
        for record in self:
            record.estado = 'encurso'
    def action_cambiar_finalizada(self):
        for record in self:
            record.estado = 'finalizada'
```
